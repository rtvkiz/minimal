  --certificate-identity-regexp='https://github.com/rtvkiz/minimal/' \

  ghcr.io/rtvkiz/minimal-httpd:latest

  --certificate-identity-regexp='https://github.com/rtvkiz/minimal/' \
  --certificate-oidc-issuer='https://token.actions.githubusercontent.com' \
  ghcr.io/rtvkiz/minimal-httpd:latest \
